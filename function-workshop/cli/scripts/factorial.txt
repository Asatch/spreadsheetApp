# FACTORIAL - Factorial of a number
# Loop sheet: counts down from N, accumulates product

new FACTORIAL loop
input NUM 5

# Row 0: initial values
write A0 =NUM
write B0 1

# Row 1: iteration body
write A1 =A0-1
write B1 =A0*B0

# Stop conditions
write _STOP0 =NUM<=0
write _STOP1 =A1<=1

# Column headers
header A counter
header B result

# Output
output B last

# Formatting
default-format number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# 0! = 1 (by definition)
test "0" "1"

# 1! = 1
test "1" "1"

# 3! = 6
test "3" "6"

# 5! = 120
test "5" "120"

# 10! = 3628800
test "10" "3628800"

save
