# SYD - Sum-of-Years-Digits Depreciation
# Accelerated depreciation: more in early periods, less later
# Formula: (Cost - Salvage) × (Remaining Life / Sum of Years)

new SYD
input COST 10000
input SALVAGE 2000
input LIFE 5
input PER 1

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'SUM-OF-YEARS DEPRECIATION
style A1 bold
style A1 fontsize 24
write A2 'Sum-of-years-digits depreciation for a specific period.
write A3 'Accelerates depreciation: more in early years, less later.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-9: THE ASSET
# ═══════════════════════════════════════════════════════════════════

write A5 '─── THE ASSET ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Cost
write B6 =COST
write C6 'Original purchase price

write A7 'Salvage Value
write B7 =SALVAGE
write C7 'Value at end of useful life

write A8 'Useful Life
write B8 =LIFE
write C8 'Total periods

write A9 'Period
write B9 =PER
write C9 'Which period to calculate

# ═══════════════════════════════════════════════════════════════════
# ROW 11-15: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A11 '─── THE CALCULATION ───
highlight A11 gray
highlight B11 gray
highlight C11 gray
highlight D11 gray
highlight E11 gray
highlight F11 gray
highlight G11 gray

write A12 'Depreciable Amount
write B12 =B6-B7
write C12 'Cost minus Salvage

write A13 'Remaining Life
write B13 =B8-B9+1
write C13 'Life - Period + 1

write A14 'Sum of Years
write B14 =B8*(B8+1)/2
write C14 'n(n+1)/2 = 1+2+3+...+n

write A15 'Period Fraction
write B15 =B13/B14
write C15 'This period's share

# ═══════════════════════════════════════════════════════════════════
# ROW 17-18: RESULT
# ═══════════════════════════════════════════════════════════════════

write A17 '─── RESULT ───
highlight A17 gray
highlight B17 gray
highlight C17 gray
highlight D17 gray
highlight E17 gray
highlight F17 gray
highlight G17 gray

write A18 'Period Depreciation
write B18 =B12*B15
write C18 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B18 SYD
output SYD

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format currency 0
format B8 number 0
format B9 number 0
format B13 number 0
format B14 number 0
format B15 percent 1

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# First year: highest depreciation (5/15 of $8000)
test "10000, 2000, 5, 1" "2666.666666666667"

# Last year: lowest depreciation (1/15 of $8000)
test "10000, 2000, 5, 5" "533.3333333333334"

# Larger asset, first year (10/55 of $25000)
test "30000, 5000, 10, 1" "4545.454545454546"

# Middle year (3/15 of $8000)
test "10000, 2000, 5, 3" "1600"

# Second year (4/15 of $8000)
test "10000, 2000, 5, 2" "2133.333333333333"

save
