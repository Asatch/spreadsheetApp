# PERMUT - Permutations
# Returns the number of ordered arrangements of k items from n
# Formula: n! / (n-k)!

new PERMUT
use FACTORIAL
input N 10
input K 3

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'PERMUTATIONS
style A1 bold
style A1 fontsize 24
write A2 'Returns the number of permutations for a given number of items.
write A3 'Order matters: choosing A,B,C is different from C,B,A.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-8: THE QUESTION
# ═══════════════════════════════════════════════════════════════════

write A5 '─── THE QUESTION ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Total Items (n)
write B6 =N
write C6 'The pool to choose from

write A7 'Items to Arrange (k)
write B7 =K
write C7 'How many positions to fill

write A8 'Items Not Chosen
write B8 =B6-B7
write C8 'n - k

# ═══════════════════════════════════════════════════════════════════
# ROW 10-12: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A10 '─── THE CALCULATION ───
highlight A10 gray
highlight B10 gray
highlight C10 gray
highlight D10 gray
highlight E10 gray
highlight F10 gray
highlight G10 gray

write A11 'n!
write B11 =FACTORIAL(B6)
write C11 'All possible orderings of n items

write A12 '(n-k)!
write B12 =FACTORIAL(B8)
write C12 'Orderings of unchosen items (divide out)

# ═══════════════════════════════════════════════════════════════════
# ROW 14-15: RESULT
# ═══════════════════════════════════════════════════════════════════

write A14 '─── RESULT ───
highlight A14 gray
highlight B14 gray
highlight C14 gray
highlight D14 gray
highlight E14 gray
highlight F14 gray
highlight G14 gray

write A15 'Permutations
write B15 =B11/B12
write C15 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B15 PERMUT
output PERMUT

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Standard: 10 arrange 3
test "10, 3" "720"

# Small: 5 arrange 2
test "5, 2" "20"

# Large: poker hand, order matters
test "52, 5" "311875200"

# Edge: arrange all items
test "5, 5" "120"

# Edge: arrange 1 item
test "10, 1" "10"

save
