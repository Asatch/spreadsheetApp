# ACCUMULATION - Multi-account pre-retirement accumulation loop
# Grows 4 account types (custom/deferred-salary, taxable, tax-deferred, tax-free)
# with annual contributions and half-year timing. Tracks taxable basis with
# inflation erosion. Matches Python calculation.py lines 297-312.

new ACCUMULATION loop

input CUSTOM_BALANCE 0
input TAXABLE_BALANCE 100000
input TAXABLE_BASIS 80000
input TAX_DEFERRED_BALANCE 500000
input TAX_FREE_BALANCE 50000
input CUSTOM_SAVINGS 0
input TAXABLE_SAVINGS 10000
input TAX_DEFERRED_SAVINGS 20000
input TAX_FREE_SAVINGS 5000
input RETURN_RATE 0.05
input INFLATION_RATE 0.025
input YEARS 10

# Row 0: initial state
# B=custom, C=taxable, D=basis, E=tax_deferred, F=tax_free, G=counter
write B0 =CUSTOM_BALANCE
write C0 =TAXABLE_BALANCE
write D0 =TAXABLE_BASIS
write E0 =TAX_DEFERRED_BALANCE
write F0 =TAX_FREE_BALANCE
write G0 0

# Row 1: one year of accumulation
# Each account: full-year growth on balance + half-year growth on contribution
# Basis: add contribution at cost (no growth), then erode by inflation
write B1 =B0*(1+RETURN_RATE)+CUSTOM_SAVINGS*((1+RETURN_RATE)^0.5)
write C1 =C0*(1+RETURN_RATE)+TAXABLE_SAVINGS*((1+RETURN_RATE)^0.5)
write D1 =(D0+TAXABLE_SAVINGS)*(1-INFLATION_RATE)
write E1 =E0*(1+RETURN_RATE)+TAX_DEFERRED_SAVINGS*((1+RETURN_RATE)^0.5)
write F1 =F0*(1+RETURN_RATE)+TAX_FREE_SAVINGS*((1+RETURN_RATE)^0.5)
write G1 =G0+1

# Stop conditions
write _STOP0 =YEARS<=0
write _STOP1 =G1>=YEARS

# Column headers
header B ENDING_CUSTOM
header C ENDING_TAXABLE
header D ENDING_BASIS
header E ENDING_TAX_DEFERRED
header F ENDING_TAX_FREE
header G YEAR

# Outputs
output B last
output C last
output D last
output E last
output F last

default-format currency 0
format G0 number 0
format G1 number 0
