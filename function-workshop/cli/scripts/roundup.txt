# ROUNDUP - Round up away from zero
# Rounds a number up (away from zero) to specified decimal places

new ROUNDUP
use ABS
use INT
input NUMBER 3.14159
input NUM_DIGITS 2

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'ROUND UP AWAY FROM ZERO
style A1 bold
style A1 fontsize 24
write A2 'Rounds a number up away from zero.
write A3 'Ceiling to specified decimal places.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-7: INPUTS
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUTS ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'The value to round

write A7 'Num Digits
write B7 =NUM_DIGITS
write C7 'Decimal places (negative = left of decimal)

# ═══════════════════════════════════════════════════════════════════
# ROW 9-17: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A9 '─── THE CALCULATION ───
highlight A9 gray
highlight B9 gray
highlight C9 gray
highlight D9 gray
highlight E9 gray
highlight F9 gray
highlight G9 gray

write A10 'Absolute Value
write B10 =ABS(B6)
write C10 'Work with positive number

write A11 'Scale Factor
write B11 =10^B7
write C11 '10 ^ num_digits

write A12 'Scaled Value
write B12 =B10*B11
write C12 'Shift decimal right

write A13 'Truncated
write B13 =INT(B12)
write C13 'Integer part

write A14 'Has Fraction?
write B14 =IF(B13<B12,1,0)
write C14 '1 if we need to round up

write A15 'Rounded Up
write B15 =B13+B14
write C15 'Add 1 if had fraction

write A16 'Unscaled
write B16 =B15/B11
write C16 'Shift decimal back

write A17 'Sign
write B17 =IF(B6<0,-1,1)
write C17 'Preserve original sign

# ═══════════════════════════════════════════════════════════════════
# ROW 19-20: RESULT
# ═══════════════════════════════════════════════════════════════════

write A19 '─── RESULT ───
highlight A19 gray
highlight B19 gray
highlight C19 gray
highlight D19 gray
highlight E19 gray
highlight F19 gray
highlight G19 gray

write A20 'Rounded Up
write B20 =B17*B16
write C20 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B20 ROUNDUP
output ROUNDUP

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 6
format B7 number 0
format B11 number 0
format B13 number 0
format B14 number 0
format B15 number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Round up to 2 decimal places
test "3.14159, 2" "3.15"

# Negative rounds away from zero
test "-3.14159, 2" "-3.15"

# Negative digits: round left of decimal
test "31415.92, -2" "31500"

# Round up to integer
test "3.1, 0" "4"

# Already exact (no change)
test "5, 0" "5"

save
