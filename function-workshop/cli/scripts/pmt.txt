# PMT - Periodic Payment for a loan
# Returns the constant payment amount for a loan with fixed rate
# Sign convention: loan amount (PV) is negative

new PMT
input RATE 0.08
input NPER 10
input PV -10000
input FV 0
input TYPE 0
default-format currency 2

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'PERIODIC PAYMENT
style A1 bold
style A1 fontsize 24
write A2 'Returns the periodic payment for a loan.
write A3 'Assumes constant payments and interest rate.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-10: INPUTS
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUTS ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Interest Rate
write B6 =RATE
write C6 'Rate per period

write A7 'Periods (n)
write B7 =NPER
write C7 'Total number of payments

write A8 'Present Value
write B8 =PV
write C8 'Loan amount

write A9 'Future Value
write B9 =FV
write C9 'Balance after last payment

write A10 'Payment Type
write B10 =TYPE
write C10 '0 = end of period, 1 = beginning

# ═══════════════════════════════════════════════════════════════════
# ROW 12-16: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A12 '─── THE CALCULATION ───
highlight A12 gray
highlight B12 gray
highlight C12 gray
highlight D12 gray
highlight E12 gray
highlight F12 gray
highlight G12 gray

write C12 'PMT = -r(PV*disc + FV) / ((disc-1)*(1+r*type))

write A13 'Discount Factor
write B13 =(1+B6)^B7
write C13 '(1+r)^n

write A14 'PV at End
write B14 =B8*B13
write C14 'PV × discount factor

write A15 'Total at End
write B15 =B14+B9
write C15 'PV growth + FV

write A16 'Numerator
write B16 =-B6*B15
write C16 '-rate × total

# ═══════════════════════════════════════════════════════════════════
# ROW 18-21: DENOMINATOR
# ═══════════════════════════════════════════════════════════════════

write A18 '─── DENOMINATOR ───
highlight A18 gray
highlight B18 gray
highlight C18 gray
highlight D18 gray
highlight E18 gray
highlight F18 gray
highlight G18 gray

write C18 'Annuity factor with timing

write A19 'Annuity Base
write B19 =B13-1
write C19 'disc - 1

write A20 'Timing Factor
write B20 =IF(B10=1,1+B6,1)
write C20 'Adjust for beginning-of-period

write A21 'Denominator
write B21 =B19*B20
write C21 'Annuity base × timing

# ═══════════════════════════════════════════════════════════════════
# ROW 23-24: RESULT
# ═══════════════════════════════════════════════════════════════════

write A23 '─── RESULT ───
highlight A23 gray
highlight B23 gray
highlight C23 gray
highlight D23 gray
highlight E23 gray
highlight F23 gray
highlight G23 gray

write A24 'Payment
write B24 =B16/B21
write C24 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B24 PMT
output PMT

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

format B6 percent 2
format B7 number 0
format B10 number 0
format B13 number 4
format B19 number 4
format B20 number 4
format B21 number 4

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Standard 10-year loan at 8%
test "0.08, 10, -10000, 0, 0" "1490.2948869707534"

# 30-year mortgage at 6%
test "0.06, 30, -200000, 0, 0" "14529.78"

# Savings target (FV with beginning-of-period)
test "0.05, 5, 0, -10000, 1" "1723.57"

# 5-year car loan at 5%
test "0.05, 5, -20000, 0, 0" "4619.495962565358"

# Loan with future balance remaining
test "0.08, 10, -10000, -5000, 0" "1835.4423304561299"

save
