# FV - Future Value of an investment
# Returns the future value based on periodic payments and constant rate
# Sign convention: investments (payments) are negative

new FV
input RATE 0.06
input NPER 10
input PMT -200
input PV -500
input TYPE 1
default-format currency 2

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'FUTURE VALUE
style A1 bold
style A1 fontsize 24
write A2 'Returns the future value of an investment.
write A3 'Based on periodic payments and constant interest rate.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-10: INPUTS
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUTS ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Interest Rate
write B6 =RATE
write C6 'Rate per period

write A7 'Periods (n)
write B7 =NPER
write C7 'Total payment periods

write A8 'Payment
write B8 =PMT
write C8 'Amount paid each period

write A9 'Present Value
write B9 =PV
write C9 'Initial investment

write A10 'Payment Type
write B10 =TYPE
write C10 '0 = end, 1 = beginning

# ═══════════════════════════════════════════════════════════════════
# ROW 12-15: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A12 '─── THE CALCULATION ───
highlight A12 gray
highlight B12 gray
highlight C12 gray
highlight D12 gray
highlight E12 gray
highlight F12 gray
highlight G12 gray

write A13 'Growth Factor
write B13 =1+B6
write C13 '1 + rate

write A14 'Compound Factor
write B14 =B13^B7
write C14 '(1+r)^n

write A15 'PV Growth
write B15 =-B9*B14
write C15 'How PV grows over time

# ═══════════════════════════════════════════════════════════════════
# ROW 17-20: PMT COMPONENT
# ═══════════════════════════════════════════════════════════════════

write A17 '─── PMT COMPONENT ───
highlight A17 gray
highlight B17 gray
highlight C17 gray
highlight D17 gray
highlight E17 gray
highlight F17 gray
highlight G17 gray

write C17 'How payments accumulate

write A18 'Annuity Factor
write B18 =(B14-1)/B6
write C18 '((1+r)^n - 1) / r

write A19 'Timing Adjustment
write B19 =IF(B10=1,B13,1)
write C19 'Extra growth for beginning-of-period

write A20 'PMT Growth
write B20 =-B8*B18*B19
write C20 'Accumulated payment value

# ═══════════════════════════════════════════════════════════════════
# ROW 22-23: RESULT
# ═══════════════════════════════════════════════════════════════════

write A22 '─── RESULT ───
highlight A22 gray
highlight B22 gray
highlight C22 gray
highlight D22 gray
highlight E22 gray
highlight F22 gray
highlight G22 gray

write A23 'Future Value
write B23 =B15+B20
write C23 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B23 FV
output FV

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

format B6 percent 2
format B7 number 0
format B10 number 0
format B13 number 4
format B14 number 4
format B18 number 4
format B19 number 4

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# $500 initial + $200/period for 10 years at 6%, beginning
test "0.06, 10, -200, -500, 1" "3689.75"

# $100/period for 5 years at 8%, end
test "0.08, 5, -100, 0, 0" "586.66"

# Lump sum only: $1000 for 3 years at 5%
test "0.05, 3, 0, -1000, 0" "1157.625"

# Periodic savings: $500/period for 10 years at 7%
test "0.07, 10, -500, 0, 0" "6908.22398063976"

# PV with end-of-period payments
test "0.06, 10, -200, -500, 0" "3531.5828367476092"

save
