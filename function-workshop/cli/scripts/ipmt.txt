# IPMT - Interest Portion of a Payment
# Shows how much of a specific payment goes to interest
# Depends on the outstanding balance at that period

new IPMT
input RATE 0.08
input PER 1
input NPER 10
input PV -10000
input FV 0
input TYPE 0
default-format currency 2

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'INTEREST PAYMENT
style A1 bold
style A1 fontsize 24
write A2 'Returns the interest portion of a payment.
write A3 'For a loan with constant payments and interest rate.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-11: INPUTS
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUTS ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Interest Rate
write B6 =RATE
write C6 'Rate per period

write A7 'Period
write B7 =PER
write C7 'Which payment (1 to nper)

write A8 'Total Periods
write B8 =NPER
write C8 'Total number of payments

write A9 'Present Value
write B9 =PV
write C9 'Loan amount

write A10 'Future Value
write B10 =FV
write C10 'Balance after last payment

write A11 'Payment Type
write B11 =TYPE
write C11 '0 = end, 1 = beginning

# ═══════════════════════════════════════════════════════════════════
# ROW 13-16: CALCULATE PAYMENT
# ═══════════════════════════════════════════════════════════════════

write A13 '─── CALCULATE PAYMENT ───
highlight A13 gray
highlight B13 gray
highlight C13 gray
highlight D13 gray
highlight E13 gray
highlight F13 gray
highlight G13 gray

write C13 'First find the periodic payment

write A14 'Discount Factor
write B14 =(1+B6)^B8
write C14 '(1+r)^n

write A15 'Timing Factor
write B15 =IF(B11=1,1+B6,1)
write C15 'Adjust for payment timing

write A16 'Payment (PMT)
write B16 =-B6*(B9*B14+B10)/((B14-1)*B15)
write C16 'Constant periodic payment

# ═══════════════════════════════════════════════════════════════════
# ROW 18-23: BALANCE AT PERIOD START
# ═══════════════════════════════════════════════════════════════════

write A18 '─── BALANCE AT PERIOD START ───
highlight A18 gray
highlight B18 gray
highlight C18 gray
highlight D18 gray
highlight E18 gray
highlight F18 gray
highlight G18 gray

write A19 'Growth to Period
write B19 =(1+B6)^(B7-1)
write C19 '(1+r)^(per-1)

write A20 'PV at Period Start
write B20 =B9*B19
write C20 'How PV has grown

write A21 'Payments Factor
write B21 =IF(B6=0,B7-1,(B19-1)/B6)
write C21 'Annuity factor for prior payments

write A22 'Prior Payments
write B22 =B16*B21*B15
write C22 'Sum of prior payments at this point

write A23 'Starting Balance
write B23 =B20+B22
write C23 'Net balance at period start

# ═══════════════════════════════════════════════════════════════════
# ROW 25-26: RESULT
# ═══════════════════════════════════════════════════════════════════

write A25 '─── RESULT ───
highlight A25 gray
highlight B25 gray
highlight C25 gray
highlight D25 gray
highlight E25 gray
highlight F25 gray
highlight G25 gray

write A26 'Interest Payment
write B26 =B23*B6
write C26 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B26 IPMT
output IPMT

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

format B6 percent 2
format B7 number 0
format B8 number 0
format B11 number 0
format B14 number 4
format B15 number 4
format B19 number 4
format B21 number 4

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# First payment: all interest on full balance
test "0.08, 1, 10, -10000, 0, 0" "-800"

# Period 5: declining interest
test "0.08, 5, 10, -10000, 0, 0" "-551.1563141025954"

# Last payment: minimal interest
test "0.08, 10, 10, -10000, 0, 0" "-110.3922138496855"

# Beginning-of-period payment
test "0.05, 1, 5, -10000, 0, 1" "-500"

# Period 3
test "0.08, 3, 10, -10000, 0, 0" "-685.1349308080665"

save
