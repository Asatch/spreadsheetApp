# TRUNC - Truncate number
# Truncates a number to a specified number of decimal places

new TRUNC
use ABS
use INT
input NUMBER 8.9
input NUM_DIGITS 0

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'TRUNCATE NUMBER
style A1 bold
style A1 fontsize 24
write A2 'Truncates a number to a specified number of digits.
write A3 'Removes fractional part, rounding toward zero.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-7: INPUTS
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUTS ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'The number to truncate

write A7 'Num Digits
write B7 =NUM_DIGITS
write C7 'Number of decimal places to keep

# ═══════════════════════════════════════════════════════════════════
# ROW 9-15: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A9 '─── THE CALCULATION ───
highlight A9 gray
highlight B9 gray
highlight C9 gray
highlight D9 gray
highlight E9 gray
highlight F9 gray
highlight G9 gray

write A10 'Is Negative?
write B10 =IF(B6<0,1,0)
write C10 'Check sign for later

write A11 'Absolute Value
write B11 =ABS(B6)
write C11 'Work with positive number

write A12 'Scale Factor
write B12 =10^B7
write C12 '10 ^ num_digits

write A13 'Scaled Value
write B13 =B11*B12
write C13 'abs(number) * scale

write A14 'Truncated Scaled
write B14 =INT(B13)
write C14 'Remove fractional part

write A15 'Unscaled Value
write B15 =B14/B12
write C15 'Divide by scale factor

# ═══════════════════════════════════════════════════════════════════
# ROW 17-18: RESULT
# ═══════════════════════════════════════════════════════════════════

write A17 '─── RESULT ───
highlight A17 gray
highlight B17 gray
highlight C17 gray
highlight D17 gray
highlight E17 gray
highlight F17 gray
highlight G17 gray

write A18 'Truncated Number
write B18 =IF(B6<0,-B15,B15)
write C18 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B18 TRUNC
output TRUNC

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 4
format B7 number 0
format B10 number 0
format B12 number 0
format B14 number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Basic truncation to integer
test "8.9, 0" "8"

# Negative truncation toward zero
test "-8.9, 0" "-8"

# Keep 1 decimal place
test "0.45, 1" "0.4"

# Negative with 2 decimal places
test "-3.567, 2" "-3.56"

# Already an integer (no change)
test "5, 0" "5"

save
