# CALCULATE_RMD - Required Minimum Distribution
# Python signature: calculate_rmd(balance, age) -> RMD amount
# Returns 0 if age < 75, otherwise balance / divisor from IRS table

new CALCULATE_RMD
input BALANCE 100000
input AGE 77

# ═══════════════════════════════════════════════════════════════════
# ROW 1-2: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'REQUIRED MINIMUM DISTRIBUTION
style A1 bold
style A1 fontsize 24
write A2 'Calculates the minimum amount you must withdraw from retirement accounts each year.

# ═══════════════════════════════════════════════════════════════════
# ROW 4-5: YOUR SITUATION
# ═══════════════════════════════════════════════════════════════════

write A4 '─── YOUR SITUATION ───
highlight A4 gray
highlight B4 gray
highlight C4 gray
highlight D4 gray
highlight E4 gray
highlight F4 gray
highlight G4 gray

write A5 'Account Balance
write B5 =BALANCE
write C5 'Total in tax-deferred accounts

write A6 'Your Age
write B6 =AGE
write C6 'As of end of year

# ═══════════════════════════════════════════════════════════════════
# RIGHT SIDE: IRS LIFE EXPECTANCY TABLE (columns E-G)
# ═══════════════════════════════════════════════════════════════════

write E4 '─── IRS TABLE ───

write G6 'Match?
write H6 'Result

# Load the RMD table (ages 72-120, row 6 = CSV header, rows 7-55 = data)
csv E6 cli/data/rmd-table.csv

# Column G: Check if this row's age >= input AGE (first match wins)
write G7 =E7>=AGE
fill G7:G55

# Column H: Cascade pattern - first match wins
write H7 =IF(G7,F7,0)
write H8 =IF(H7>0,H7,IF(G8,F8,0))
fill H8:H55

# Named range for the final cascade result (referenced 46 rows away)
name H55 DIVISOR_LOOKUP

# ═══════════════════════════════════════════════════════════════════
# ROW 8-11: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE CALCULATION ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray

write A9 'Your Divisor
write B9 =DIVISOR_LOOKUP
write C9 'From IRS table for your age

write A10 'RMD Required?
write B10 =AGE>=75
write C10 'RMDs begin at age 75

write A11 'RMD Amount
write B11 =IF(B10,BALANCE/B9,0)
write C11 'Balance ÷ Divisor (or 0 if under 75)

# ═══════════════════════════════════════════════════════════════════
# ROW 13: RESULT
# ═══════════════════════════════════════════════════════════════════

write A13 '─── RESULT ───
highlight A13 gray
highlight B13 gray
highlight C13 gray
highlight D13 gray

write A14 'Your RMD
write B14 =B11
write C14 '← Amount you must withdraw this year

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B14 RMD
output RMD
