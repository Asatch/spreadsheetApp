# RETIREMENT_YEAR - One complete year of retirement
# Custom account distributions, income, taxes, withdrawals, investment growth,
# and terminal value — everything in one function.
# Returns: ending balances (taxable, basis, tax-deferred, tax-free, custom),
#          total taxes, cumulative taxes, cumulative shortfall, terminal value
# Output names: ENDING_TAXABLE, ENDING_BASIS, ENDING_TAX_DEFERRED, ENDING_TAX_FREE,
#               ENDING_CUSTOM, TOTAL_TAXES, CUM_TAXES_END, CUM_SHORTFALL_END, TERMINAL_VALUE

new RETIREMENT_YEAR

use CUSTOM_ACCOUNT_DIST
use CALCULATE_RMD
use PROGRESSIVE_TAX
use PAY_FOR_EXPENSES

input AGE 77
input TAXABLE_BALANCE 100000
input TAXABLE_BASIS 80000
input TAX_DEFERRED_BALANCE 500000
input TAX_FREE_BALANCE 50000
input CUSTOM_BALANCE 0
input YEAR 1
input CUM_TAXES 0
input CUM_SHORTFALL 0

input SS_AGE 67
input SS_INCOME 30000
input SPENDING 60000
input RETURN_RATE 0.05
input INFLATION_RATE 0.025
input TERMINAL_TAX_RATE 0.25
input STRATEGY 1

# ═══════════════════════════════════════════════════════════════════
# ROW 1-2: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'RETIREMENT YEAR
style A1 bold
style A1 fontsize 24
write A2 'One complete year of retirement: custom distributions, income, taxes, withdrawals, growth, terminal value.

# ═══════════════════════════════════════════════════════════════════
# ROW 4-8: CUSTOM ACCOUNT
# ═══════════════════════════════════════════════════════════════════

write A4 '─── CUSTOM ACCOUNT ───
highlight A4 gray
highlight B4 gray
highlight C4 gray
highlight D4 gray
highlight E4 gray
highlight F4 gray
highlight G4 gray

write A5 'Starting Balance
write B5 =CUSTOM_BALANCE

write A6 'Mid-Year Balance
write B6 =CUSTOM_BALANCE*((1+RETURN_RATE)^0.5)
write C6 'Half-year growth before distribution

write A7 'Distribution
write B7 =CUSTOM_ACCOUNT_DIST(B6, YEAR)
write C7 'Scheduled annual distribution

write A8 'Ending Balance
write B8 =(B6-B7)*((1+RETURN_RATE)^0.5)
write C8 'Second half-year growth after distribution

# ═══════════════════════════════════════════════════════════════════
# ROW 10-16: INCOME SOURCES
# ═══════════════════════════════════════════════════════════════════

write A10 '─── INCOME SOURCES ───
highlight A10 gray
highlight B10 gray
highlight C10 gray
highlight D10 gray
highlight E10 gray
highlight F10 gray
highlight G10 gray

write A11 'Age
write B11 =AGE
format B11 number 0

write A12 'Custom Income
write B12 =B7
write C12 'Distribution from custom account

write A13 'Social Security
write B13 =IF(AGE>=SS_AGE, SS_INCOME, 0)
write C13 'SS if at or past SS age

write A14 'RMD from Tax-Deferred
write B14 =CALCULATE_RMD(TAX_DEFERRED_BALANCE, AGE)
write C14 'Required minimum distribution

write A15 'Tax-Def After RMD
write B15 =TAX_DEFERRED_BALANCE-B14
write C15 'Tax-deferred balance minus RMD

write A16 'Total Income
write B16 =B12+B13+B14
write C16 'Custom + SS + RMD

# ═══════════════════════════════════════════════════════════════════
# ROW 18-20: TAX ON INCOME
# ═══════════════════════════════════════════════════════════════════

write A18 '─── TAX ON INCOME ───
highlight A18 gray
highlight B18 gray
highlight C18 gray
highlight D18 gray
highlight E18 gray
highlight F18 gray
highlight G18 gray

write A19 'Income Tax
write B19 =PROGRESSIVE_TAX(B16)
write C19 'Federal income tax on total income

write A20 'After-Tax Income
write B20 =B16-B19
write C20 'Total income minus income tax

# ═══════════════════════════════════════════════════════════════════
# ROW 22-25: SPENDING ANALYSIS
# ═══════════════════════════════════════════════════════════════════

write A22 '─── SPENDING ANALYSIS ───
highlight A22 gray
highlight B22 gray
highlight C22 gray
highlight D22 gray
highlight E22 gray
highlight F22 gray
highlight G22 gray

write A23 'Annual Spending
write B23 =SPENDING
write C23 'Annual expenses in retirement

write A24 'Shortfall
write B24 =MAX(0, B23-B20)
write C24 'Amount needing withdrawal

write A25 'Savings
write B25 =MAX(0, B20-B23)
write C25 'Surplus going to taxable

# ═══════════════════════════════════════════════════════════════════
# ROW 27-36: WITHDRAWALS
# ═══════════════════════════════════════════════════════════════════

write A27 '─── WITHDRAWALS ───
highlight A27 gray
highlight B27 gray
highlight C27 gray
highlight D27 gray
highlight E27 gray
highlight F27 gray
highlight G27 gray

write A28 'Strategy
write B28 =STRATEGY
write C28 '1=Taxable, 2=Tax-Def, 3=Tax-Free first
format B28 number 0

write A29 'Expenses Result
write B29 =PAY_FOR_EXPENSES(TAXABLE_BALANCE, TAXABLE_BASIS, B15, TAX_FREE_BALANCE, B24, B16, STRATEGY)
write C29 'PAY_FOR_EXPENSES result (see below)

write A31 'Taxable After
write B31 =INDEX(B29, "ENDING_TAXABLE")
write C31 'Taxable balance after withdrawals

write A32 'Basis After
write B32 =INDEX(B29, "ENDING_BASIS")
write C32 'Cost basis after withdrawals

write A33 'Tax-Def After
write B33 =INDEX(B29, "ENDING_TAX_DEFERRED")
write C33 'Tax-deferred balance after withdrawals

write A34 'Tax-Free After
write B34 =INDEX(B29, "ENDING_TAX_FREE")
write C34 'Tax-free balance after withdrawals

write A35 'Withdrawal Tax
write B35 =INDEX(B29, "WITHDRAWAL_TAX")
write C35 'Tax from capital gains on withdrawals

write A36 'Remaining Shortfall
write B36 =INDEX(B29, "REMAINING_SHORTFALL")
write C36 'Unfunded spending if accounts exhausted

# ═══════════════════════════════════════════════════════════════════
# ROW 38-43: INVESTMENT RETURNS
# ═══════════════════════════════════════════════════════════════════

write A38 '─── INVESTMENT RETURNS ───
highlight A38 gray
highlight B38 gray
highlight C38 gray
highlight D38 gray
highlight E38 gray
highlight F38 gray
highlight G38 gray
write C38 'Mid-year avg: return × (start + end) / 2

write A39 'Taxable + Savings
write B39 =IF(B24>0, B31, TAXABLE_BALANCE+B25)
write C39 'If shortfall: post-withdrawal. If savings: add to taxable

write A40 'Basis + Savings
write B40 =IF(B24>0, B32, TAXABLE_BASIS+B25)
write C40 'If shortfall: post-withdrawal. If savings: add to basis

write A41 'Taxable End
write B41 =B39+RETURN_RATE*(TAXABLE_BALANCE+B39)/2
write C41 'After mid-year average investment growth

write A42 'Tax-Def End
write B42 =B33+RETURN_RATE*(B15+B33)/2
write C42 'Start = after RMD, end = after withdrawals

write A43 'Tax-Free End
write B43 =B34+RETURN_RATE*(TAX_FREE_BALANCE+B34)/2
write C43 'After mid-year average investment growth

# ═══════════════════════════════════════════════════════════════════
# ROW 45-55: RESULT
# ═══════════════════════════════════════════════════════════════════

write A45 '─── RESULT ───
highlight A45 gray
highlight B45 gray
highlight C45 gray
highlight D45 gray
highlight E45 gray
highlight F45 gray
highlight G45 gray

write A46 'Ending Taxable
write B46 =B41
write C46 '← OUTPUT

write A47 'Ending Basis
write B47 =B40*(1-INFLATION_RATE)
write C47 '← OUTPUT: Basis eroded by inflation

write A48 'Ending Tax-Deferred
write B48 =B42
write C48 '← OUTPUT

write A49 'Ending Tax-Free
write B49 =B43
write C49 '← OUTPUT

write A50 'Ending Custom
write B50 =B8
write C50 '← OUTPUT

write A51 'Year Taxes
write B51 =B19+B35
write C51 'Income tax + withdrawal tax

write A52 'Cumulative Taxes
write B52 =CUM_TAXES+B51
write C52 '← OUTPUT: Running total of actual taxes paid

write A53 'Total Taxes
write B53 =B52+TERMINAL_TAX_RATE*B48
write C53 '← OUTPUT: Plus terminal tax on tax-deferred

write A54 'Cumulative Shortfall
write B54 =CUM_SHORTFALL+B36
write C54 '← OUTPUT

write A55 'Terminal Value
write B55 =B46+B50+B48*(1-TERMINAL_TAX_RATE)+B49
write C55 '← OUTPUT: Tax-adjusted net worth if liquidated now

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B46 ENDING_TAXABLE
name B47 ENDING_BASIS
name B48 ENDING_TAX_DEFERRED
name B49 ENDING_TAX_FREE
name B50 ENDING_CUSTOM
name B53 TOTAL_TAXES
name B52 CUM_TAXES_END
name B54 CUM_SHORTFALL_END
name B55 TERMINAL_VALUE

output ENDING_TAXABLE
output ENDING_BASIS
output ENDING_TAX_DEFERRED
output ENDING_TAX_FREE
output ENDING_CUSTOM
output TOTAL_TAXES
output CUM_TAXES_END
output CUM_SHORTFALL_END
output TERMINAL_VALUE

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format currency 0
format RETURN_RATE percent 1
format INFLATION_RATE percent 1
format TERMINAL_TAX_RATE percent 0
