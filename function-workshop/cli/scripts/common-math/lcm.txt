# LCM - Least Common Multiple
# LCM(a, b) = |a * b| / GCD(a, b)

new LCM
input NUM1 12
input NUM2 8

use ABS
use GCD

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'LEAST COMMON MULTIPLE
style A1 bold
style A1 fontsize 24
write A2 'Returns the smallest positive integer divisible by both inputs.
write A3 'Computed as |a * b| / GCD(a, b).

# ═══════════════════════════════════════════════════════════════════
# ROW 5-7: INPUTS
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUTS ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number 1
write B6 =NUM1
write C6 'First number

write A7 'Number 2
write B7 =NUM2
write C7 'Second number

# ═══════════════════════════════════════════════════════════════════
# ROW 9-12: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A9 '─── THE CALCULATION ───
highlight A9 gray
highlight B9 gray
highlight C9 gray
highlight D9 gray
highlight E9 gray
highlight F9 gray
highlight G9 gray

write A10 '|A * B|
write B10 =ABS(B6*B7)
write C10 'Absolute value of the product

write A11 'GCD(A, B)
write B11 =GCD(B6,B7)
write C11 'Greatest common divisor

# ═══════════════════════════════════════════════════════════════════
# ROW 13-14: RESULT
# ═══════════════════════════════════════════════════════════════════

write A13 '─── RESULT ───
highlight A13 gray
highlight B13 gray
highlight C13 gray
highlight D13 gray
highlight E13 gray
highlight F13 gray
highlight G13 gray

write A14 'LCM
write B14 =IF(B11=0,0,B10/B11)
write C14 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B14 LCM
output LCM

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Classic example
test "12, 8" "24"

# Coprime
test "7, 5" "35"

# One divides the other
test "6, 3" "6"

# Same number
test "9, 9" "9"

# One is zero
test "5, 0" "0"

save
