# COS - Cosine
# Returns cosine using the identity COS(x) = SIN(x + pi/2)

new COS
input X 0

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'COSINE
style A1 bold
style A1 fontsize 24
write A2 'Returns the cosine of an angle in radians.
write A3 'Uses the identity: COS(x) = SIN(x + pi/2).

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'X
write B6 =X
write C6 'Angle in radians

# ═══════════════════════════════════════════════════════════════════
# ROW 8-10: THE IDENTITY
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE IDENTITY ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'pi / 2
write B9 =1.5707963267948966
write C9 '90 degrees in radians

write A10 'Shifted Angle
write B10 =B6+B9
write C10 'x + pi/2

# ═══════════════════════════════════════════════════════════════════
# ROW 12-13: RESULT
# ═══════════════════════════════════════════════════════════════════

write A12 '─── RESULT ───
highlight A12 gray
highlight B12 gray
highlight C12 gray
highlight D12 gray
highlight E12 gray
highlight F12 gray
highlight G12 gray

write A13 'Cosine
write B13 =SIN(B10)
write C13 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B13 COS
output COS

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 6

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# cos(0) = 1
test "0" "1"

# cos(pi/2) = 0
test "1.5707963267948966" "0"

# cos(pi) = -1
test "3.141592653589793" "-1"

# cos(pi/4) = sqrt(2)/2
test "0.7853981633974483" "0.707107"

# cos(2*pi) = 1
test "6.283185307179586" "1"

save
