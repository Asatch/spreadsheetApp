# TAN - Tangent
# Returns tangent as SIN(x) / COS(x)

new TAN
input X 0

use COS

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'TANGENT
style A1 bold
style A1 fontsize 24
write A2 'Returns the tangent of an angle in radians.
write A3 'Computed as SIN(x) / COS(x).

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'X
write B6 =X
write C6 'Angle in radians

# ═══════════════════════════════════════════════════════════════════
# ROW 8-10: THE RATIO
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE RATIO ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'sin(x)
write B9 =SIN(B6)
write C9 'Opposite / Hypotenuse

write A10 'cos(x)
write B10 =COS(B6)
write C10 'Adjacent / Hypotenuse

# ═══════════════════════════════════════════════════════════════════
# ROW 12-13: RESULT
# ═══════════════════════════════════════════════════════════════════

write A12 '─── RESULT ───
highlight A12 gray
highlight B12 gray
highlight C12 gray
highlight D12 gray
highlight E12 gray
highlight F12 gray
highlight G12 gray

write A13 'Tangent
write B13 =B9/B10
write C13 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B13 TAN
output TAN

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 6

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# tan(0) = 0
test "0" "0"

# tan(pi/4) = 1
test "0.7853981633974483" "1"

# tan(pi/6) = 1/sqrt(3)
test "0.5235987755982988" "0.577350"

# tan(pi) = 0
test "3.141592653589793" "0"

# tan(-pi/4) = -1
test "-0.7853981633974483" "-1"

save
