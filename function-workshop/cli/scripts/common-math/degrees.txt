# DEGREES - Convert radians to degrees
# degrees = radians * 180 / pi

new DEGREES
input RADIANS 3.141592653589793

use PI

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'DEGREES
style A1 bold
style A1 fontsize 24
write A2 'Converts radians to degrees.
write A3 'Multiply by 180/pi.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Radians
write B6 =RADIANS
write C6 'Angle in radians

# ═══════════════════════════════════════════════════════════════════
# ROW 8-10: THE CONVERSION
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE CONVERSION ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'pi
write B9 =PI(1)
write C9 '3.14159...

write A10 '180 / pi
write B10 =180/B9
write C10 'Conversion factor

# ═══════════════════════════════════════════════════════════════════
# ROW 12-13: RESULT
# ═══════════════════════════════════════════════════════════════════

write A12 '─── RESULT ───
highlight A12 gray
highlight B12 gray
highlight C12 gray
highlight D12 gray
highlight E12 gray
highlight F12 gray
highlight G12 gray

write A13 'Degrees
write B13 =B6*B10
write C13 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B13 DEGREES
output DEGREES

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 6

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# pi = 180°
test "3.141592653589793" "180"

# pi/2 = 90°
test "1.5707963267948966" "90"

# 0
test "0" "0"

# 2*pi = 360°
test "6.283185307179586" "360"

# 1 radian ≈ 57.2958°
test "1" "57.295780"

save
