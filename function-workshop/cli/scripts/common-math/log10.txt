# LOG10 - Base-10 logarithm
# Returns the base-10 logarithm of a number

new LOG10
input NUMBER 100

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'BASE-10 LOGARITHM
style A1 bold
style A1 fontsize 24
write A2 'Returns the base-10 logarithm of a number.
write A3 'Answers: 10 to what power equals this number?

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'Must be positive

# ═══════════════════════════════════════════════════════════════════
# ROW 8-10: CHANGE OF BASE
# ═══════════════════════════════════════════════════════════════════

write A8 '─── CHANGE OF BASE ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'ln(x)
write B9 =LN(B6)
write C9 'Natural logarithm

write A10 'ln(10)
write B10 =LN(10)
write C10 'Constant: 2.302585...

# ═══════════════════════════════════════════════════════════════════
# ROW 12-13: RESULT
# ═══════════════════════════════════════════════════════════════════

write A12 '─── RESULT ───
highlight A12 gray
highlight B12 gray
highlight C12 gray
highlight D12 gray
highlight E12 gray
highlight F12 gray
highlight G12 gray

write A13 'Log Base 10
write B13 =B9/B10
write C13 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B13 LOG10
output LOG10

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 6
format B6 number 0
format B13 number 4

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# 10^2 = 100
test "100" "2"

# 10^3 = 1000
test "1000" "3"

# 10^0 = 1
test "1" "0"

# 10^1 = 10
test "10" "1"

# 10^-1 = 0.1
test "0.1" "-1"

save
