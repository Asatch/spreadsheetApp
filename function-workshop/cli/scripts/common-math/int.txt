# INT - Integer part of a number
# Returns the integer part by rounding down toward negative infinity

new INT
input NUMBER 7.8

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'INTEGER PART
style A1 bold
style A1 fontsize 24
write A2 'Returns the integer part of a number.
write A3 'Rounds down toward negative infinity (like FLOOR).

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'The value to truncate to integer

# ═══════════════════════════════════════════════════════════════════
# ROW 8-9: RESULT
# ═══════════════════════════════════════════════════════════════════

write A8 '─── RESULT ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'Integer Part
write B9 =FLOOR(B6)
write C9 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B9 INT
output INT

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 4
format B9 number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Positive decimal
test "7.8" "7"

# Positive integer (unchanged)
test "5" "5"

# Zero
test "0" "0"

# Negative decimal (rounds toward -infinity)
test "-3.2" "-4"

# Negative integer (unchanged)
test "-5" "-5"

save
