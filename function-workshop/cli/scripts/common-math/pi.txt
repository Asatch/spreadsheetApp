# PI - Returns pi times a multiplier
# PI(1) = 3.14159..., PI(2) = 6.28318..., etc.

new PI
input MULTIPLIER 1

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'PI
style A1 bold
style A1 fontsize 24
write A2 'Returns pi times a multiplier.
write A3 'The ratio of a circle's circumference to its diameter.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Multiplier
write B6 =MULTIPLIER
write C6 'Multiply pi by this value

# ═══════════════════════════════════════════════════════════════════
# ROW 8-10: THE VALUE
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE VALUE ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'pi
write B9 =3.141592653589793
write C9 '3.14159265...

# ═══════════════════════════════════════════════════════════════════
# ROW 11-12: RESULT
# ═══════════════════════════════════════════════════════════════════

write A11 '─── RESULT ───
highlight A11 gray
highlight B11 gray
highlight C11 gray
highlight D11 gray
highlight E11 gray
highlight F11 gray
highlight G11 gray

write A12 'Result
write B12 =B9*B6
write C12 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B12 PI
output PI

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 15

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# pi * 1
test "1" "3.141592653589793"

# pi * 2
test "2" "6.283185307179586"

# pi * 0.5 (pi/2)
test "0.5" "1.5707963267948966"

# pi * 0
test "0" "0"

# pi * -1
test "-1" "-3.141592653589793"

save
