# SQRT - Square root
# Returns the square root via exponentiation

new SQRT
input NUMBER 9

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'SQUARE ROOT
style A1 bold
style A1 fontsize 24
write A2 'Returns the square root of a number.
write A3 'Equivalent to raising to the power of 0.5.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'Must be non-negative

# ═══════════════════════════════════════════════════════════════════
# ROW 8-9: RESULT
# ═══════════════════════════════════════════════════════════════════

write A8 '─── RESULT ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'Square Root
write B9 =B6^0.5
write C9 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B9 SQRT
output SQRT

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 6

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Perfect square
test "9" "3"

# Perfect square
test "4" "2"

# Zero
test "0" "0"

# One
test "1" "1"

# Irrational result
test "2" "1.414214"

# Large perfect square
test "100" "10"

# Fractional input
test "0.25" "0.5"

save
