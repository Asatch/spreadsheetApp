# PV - Present Value of an investment
# How much a future stream of payments is worth today
# Sign convention: payments received are negative

new PV
input RATE 0.08
input NPER 20
input PMT -500
input FV 0
input TYPE 0
default-format currency 2

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'PRESENT VALUE
style A1 bold
style A1 fontsize 24
write A2 'Returns the present value of an investment.
write A3 'How much a future stream of payments is worth today.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-10: INPUTS
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUTS ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Interest Rate
write B6 =RATE
write C6 'Discount rate per period

write A7 'Periods (n)
write B7 =NPER
write C7 'Total payment periods

write A8 'Payment
write B8 =PMT
write C8 'Amount received each period

write A9 'Future Value
write B9 =FV
write C9 'Lump sum at end

write A10 'Payment Type
write B10 =TYPE
write C10 '0 = end of period, 1 = beginning

# ═══════════════════════════════════════════════════════════════════
# ROW 12-14: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A12 '─── THE CALCULATION ───
highlight A12 gray
highlight B12 gray
highlight C12 gray
highlight D12 gray
highlight E12 gray
highlight F12 gray
highlight G12 gray

write C12 'Discount future cash flows to present

write A13 'Discount Factor
write B13 =(1+B6)^B7
write C13 '(1+r)^n

write A14 'FV Discounted
write B14 =-B9/B13
write C14 'Future lump sum in today's dollars

# ═══════════════════════════════════════════════════════════════════
# ROW 16-19: PMT COMPONENT
# ═══════════════════════════════════════════════════════════════════

write A16 '─── PMT COMPONENT ───
highlight A16 gray
highlight B16 gray
highlight C16 gray
highlight D16 gray
highlight E16 gray
highlight F16 gray
highlight G16 gray

write C16 'Present value of payment stream

write A17 'Annuity Factor
write B17 =(1-1/B13)/B6
write C17 '(1 - 1/(1+r)^n) / r

write A18 'Timing Adjustment
write B18 =IF(B10=1,1+B6,1)
write C18 'Extra value for beginning-of-period

write A19 'PMT Discounted
write B19 =-B8*B17*B18
write C19 'Payments in today's dollars

# ═══════════════════════════════════════════════════════════════════
# ROW 21-22: RESULT
# ═══════════════════════════════════════════════════════════════════

write A21 '─── RESULT ───
highlight A21 gray
highlight B21 gray
highlight C21 gray
highlight D21 gray
highlight E21 gray
highlight F21 gray
highlight G21 gray

write A22 'Present Value
write B22 =B14+B19
write C22 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B22 PV
output PV

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

format B6 percent 2
format B7 number 0
format B10 number 0
format B13 number 4
format B17 number 4
format B18 number 4

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# 20-year annuity at 8%
test "0.08, 20, -500, 0, 0" "4909.07"

# Lump sum only: $10k in 10 years at 5%
test "0.05, 10, 0, -10000, 0" "6139.13"

# Annuity + lump sum with beginning-of-period
test "0.06, 5, -1000, -5000, 1" "8201.40"

# Simple 10-year annuity at 10%
test "0.10, 10, -1000, 0, 0" "6144.57"

# Short-term annuity at 3%
test "0.03, 5, -2000, 0, 0" "9159.41"

save
