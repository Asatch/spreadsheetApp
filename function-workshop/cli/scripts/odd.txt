# ODD - Round to nearest odd integer
# Rounds away from zero to the nearest odd integer

new ODD
use ABS
use INT
input NUMBER 1.5

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'ROUND TO ODD INTEGER
style A1 bold
style A1 fontsize 24
write A2 'Rounds a number away from zero to the nearest odd integer.
write A3 'Returns the next odd integer in the direction away from zero.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'The value to round

# ═══════════════════════════════════════════════════════════════════
# ROW 8-15: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE CALCULATION ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'Absolute Value
write B9 =ABS(B6)
write C9 'Work with positive number

write A10 'Ceiling
write B10 =-INT(-B9)
write C10 'Round up: CEILING = -INT(-x)

write A11 'Half Ceiling
write B11 =INT(B10/2)
write C11 'INT(ceiling / 2)

write A12 'Mod 2
write B12 =B10-B11*2
write C12 'ceiling - INT(ceiling/2)*2

write A13 'Is Even?
write B13 =IF(B12=0,1,0)
write C13 '1 if mod=0, else 0

write A14 'Odd Absolute
write B14 =IF(B12=0,B10+1,B10)
write C14 'Add 1 if even

write A15 'Sign
write B15 =IF(B6<0,-1,1)
write C15 '-1 if negative, else 1

# ═══════════════════════════════════════════════════════════════════
# ROW 17-18: RESULT
# ═══════════════════════════════════════════════════════════════════

write A17 '─── RESULT ───
highlight A17 gray
highlight B17 gray
highlight C17 gray
highlight D17 gray
highlight E17 gray
highlight F17 gray
highlight G17 gray

write A18 'Odd Integer
write B18 =IF(B6=0,1,B15*B14)
write C18 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B18 ODD
output ODD

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 2
format B10 number 0
format B11 number 0
format B12 number 0
format B13 number 0
format B14 number 0
format B18 number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Fractional rounds up to next odd
test "1.5" "3"

# Already odd (unchanged)
test "3" "3"

# Negative rounds away from zero
test "-1" "-1"

# Even rounds up to next odd
test "2" "3"

# Zero returns 1
test "0" "1"

# Negative decimal rounds away from zero
test "-4.5" "-5"

save
