# TEST_OUTPUT_ONLY_COL - Minimal reproduction of output-only loop column bug
# Column C is an output but doesn't feed back as state (C1 doesn't reference C0).
# The transpiler drops it from loop state, then the return references an undefined variable.

new TEST_OUTPUT_ONLY_COL loop

input START 100
input RATE 0.1
input YEARS 3

# Row 0: initial state
write A0 =START
write B0 0
write C0 0

# Row 1: iteration
write A1 =A0*(1+RATE)
write B1 =B0+1
write C1 =A1*2

write _STOP1 =B1>=YEARS

header A ENDING_BALANCE
header C DOUBLED

default-format number 2
format B0 number 0
format B1 number 0

output A last
output C last

# 100 * 1.1 = 110, * 1.1 = 121, * 1.1 = 133.1
# doubled = 133.1 * 2 = 266.2
test "100, 0.1, 3" "133.1, 266.2"

# 1 iteration: 50 * 1.2 = 60, doubled = 120
test "50, 0.2, 1" "60, 120"

save output/TEST_OUTPUT_ONLY_COL.xml
