# ABS - Absolute value
# Returns the absolute value of a number

new ABS
input NUMBER -5

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'ABSOLUTE VALUE
style A1 bold
style A1 fontsize 24
write A2 'Returns the absolute value of a number.
write A3 'Distance from zero, always positive (or zero).

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'The value to make positive

# ═══════════════════════════════════════════════════════════════════
# ROW 8-9: THE LOGIC
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE LOGIC ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'Is Negative?
write B9 =IF(B6<0,1,0)
write C9 'Check if less than zero

# ═══════════════════════════════════════════════════════════════════
# ROW 11-12: RESULT
# ═══════════════════════════════════════════════════════════════════

write A11 '─── RESULT ───
highlight A11 gray
highlight B11 gray
highlight C11 gray
highlight D11 gray
highlight E11 gray
highlight F11 gray
highlight G11 gray

write A12 'Absolute Value
write B12 =IF(B6<0,-B6,B6)
write C12 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B12 ABS
output ABS

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 4
format B9 number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Negative integer
test "-5" "5"

# Positive integer (unchanged)
test "5" "5"

# Zero
test "0" "0"

# Negative decimal
test "-3.14159" "3.14159"

# Large positive (unchanged)
test "1000000" "1000000"

save
