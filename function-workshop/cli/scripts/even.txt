# EVEN - Round to nearest even integer
# Rounds away from zero to the nearest even integer

new EVEN
use ABS
use INT
input NUMBER 1.5

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'ROUND TO EVEN INTEGER
style A1 bold
style A1 fontsize 24
write A2 'Rounds a number away from zero to the nearest even integer.
write A3 'Returns the next even integer in the direction away from zero.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Number
write B6 =NUMBER
write C6 'The value to round

# ═══════════════════════════════════════════════════════════════════
# ROW 8-13: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE CALCULATION ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 'Absolute Value
write B9 =ABS(B6)
write C9 'Work with positive number

write A10 'Half
write B10 =B9/2
write C10 'Divide by 2

write A11 'Ceiling of Half
write B11 =-INT(-B10)
write C11 'CEILING = -INT(-x)

write A12 'Even Absolute
write B12 =B11*2
write C12 'Multiply back by 2

write A13 'Sign
write B13 =IF(B6<0,-1,1)
write C13 '-1 if negative, else 1

# ═══════════════════════════════════════════════════════════════════
# ROW 15-16: RESULT
# ═══════════════════════════════════════════════════════════════════

write A15 '─── RESULT ───
highlight A15 gray
highlight B15 gray
highlight C15 gray
highlight D15 gray
highlight E15 gray
highlight F15 gray
highlight G15 gray

write A16 'Even Integer
write B16 =B13*B12
write C16 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B16 EVEN
output EVEN

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 2
format B11 number 0
format B12 number 0
format B13 number 0
format B16 number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Fractional rounds up to next even
test "1.5" "2"

# Odd rounds up to next even
test "3" "4"

# Negative rounds away from zero
test "-1" "-2"

# Already even (unchanged)
test "2" "2"

# Zero stays zero
test "0" "0"

save
