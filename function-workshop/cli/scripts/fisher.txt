# FISHER - Fisher z-transformation
# Transforms a correlation coefficient for statistical testing
# Formula: z = 0.5 × ln((1+r)/(1-r))

new FISHER
input X 0.75

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'FISHER TRANSFORMATION
style A1 bold
style A1 fontsize 24
write A2 'Transforms a correlation coefficient for statistical testing.
write A3 'Fisher z = 0.5 × ln((1+r)/(1-r))

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Correlation (r)
write B6 =X
write C6 'Value between -1 and 1

# ═══════════════════════════════════════════════════════════════════
# ROW 8-12: THE TRANSFORMATION
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE TRANSFORMATION ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 '1 + r
write B9 =1+B6
write C9 'Distance from -1

write A10 '1 - r
write B10 =1-B6
write C10 'Distance from +1

write A11 '(1+r)/(1-r)
write B11 =B9/B10
write C11 'Ratio of distances

write A12 'ln(ratio)
write B12 =LN(B11)
write C12 'Logarithmic scale

# ═══════════════════════════════════════════════════════════════════
# ROW 14-15: RESULT
# ═══════════════════════════════════════════════════════════════════

write A14 '─── RESULT ───
highlight A14 gray
highlight B14 gray
highlight C14 gray
highlight D14 gray
highlight E14 gray
highlight F14 gray
highlight G14 gray

write A15 'Fisher z
write B15 =0.5*B12
write C15 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B15 FISHER
output FISHER

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 4

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Moderate positive correlation
test "0.75" "0.9729550745276566"

# Zero correlation (identity)
test "0" "0"

# Moderate correlation
test "0.5" "0.5493061443340548"

# Negative correlation
test "-0.3" "-0.30951960420311175"

# Strong positive correlation (near boundary)
test "0.9" "1.4722194895832204"

save
