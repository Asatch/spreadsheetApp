# RETIREMENT_YEAR_FULL - Single year of retirement calculation
# Matches Python run_retirement_projection() single-year logic
# Uses PAY_FOR_EXPENSES_V2 for strategy-selectable withdrawal ordering

new RETIREMENT_YEAR_FULL

# Dependencies
use CALCULATE_RMD
use PROGRESSIVE_TAX
use PAY_FOR_EXPENSES_V2

# Inputs
input AGE 77
input TAXABLE_BALANCE 100000
input TAXABLE_BASIS 80000
input TAX_DEFERRED_BALANCE 500000
input TAX_FREE_BALANCE 50000
input SS_AGE 67
input SS_INCOME 30000
input SPENDING 60000
input RETURN_RATE 0.05
input INFLATION_RATE 0.025
input CUSTOM_INCOME 0
input STRATEGY 1

# Row 1-2: Docstring
write A1 'RETIREMENT YEAR (FULL MODEL)
write A2 'One year of retirement: income, taxes, withdrawals, investment growth

# Row 4: Section - Income Sources
write A4 '─── INCOME SOURCES ───

write A5 'Age This Year
write B5 =AGE
format B5 number 0

write A6 'Custom Account Income
write B6 =CUSTOM_INCOME
write C6 'Distribution from custom account

write A7 'Social Security
write B7 =IF(AGE>=SS_AGE, SS_INCOME, 0)
write C7 'SS if at or past SS age

write A8 'RMD from Tax-Deferred
write B8 =CALCULATE_RMD(TAX_DEFERRED_BALANCE, AGE)
write C8 'Required minimum distribution

write A9 'Tax-Def After RMD
write B9 =TAX_DEFERRED_BALANCE-B8

write A10 'Total Income
write B10 =B6+B7+B8

# Row 12: Section - Tax
write A12 '─── TAX ON INCOME ───

write A13 'Income Tax
write B13 =PROGRESSIVE_TAX(B10)

write A14 'After-Tax Income
write B14 =B10-B13

# Row 16: Spending Analysis
write A16 '─── SPENDING ANALYSIS ───

write A17 'Annual Spending
write B17 =SPENDING

write A18 'Shortfall
write B18 =MAX(0, B17-B14)

write A19 'Savings
write B19 =MAX(0, B14-B17)

# Row 21: Withdrawal via PAY_FOR_EXPENSES_V2
write A21 '─── WITHDRAWALS ───
write A22 'Strategy
write B22 =STRATEGY
write C22 '1=Taxable, 2=Tax-Def, 3=Tax-Free first
format B22 number 0

# Call PAY_FOR_EXPENSES_V2 with post-RMD tax-deferred balance
write A24 'PAY_FOR_EXPENSES result
write B24 =PAY_FOR_EXPENSES_V2(TAXABLE_BALANCE, TAXABLE_BASIS, B9, TAX_FREE_BALANCE, B18, B10, STRATEGY)

write A26 'Taxable After
write B26 =INDEX(B24, 1)

write A27 'Basis After
write B27 =INDEX(B24, 2)

write A28 'Tax-Def After
write B28 =INDEX(B24, 3)

write A29 'Tax-Free After
write B29 =INDEX(B24, 4)

write A30 'Withdrawal Tax
write B30 =INDEX(B24, 5)

write A31 'Remaining Shortfall
write B31 =INDEX(B24, 6)

# Row 33: Investment Growth (mid-year average method)
# Growth = return × (balance_before + balance_after) / 2
write A33 '─── INVESTMENT RETURNS ───

write A34 'Taxable + Savings
write B34 =IF(B18>0, B26, TAXABLE_BALANCE+B19)
write C34 'If shortfall: post-withdrawal. If savings: add to taxable.

write A35 'Basis + Savings
write B35 =IF(B18>0, B27, TAXABLE_BASIS+B19)

write A36 'Taxable End
write B36 =B34+RETURN_RATE*(TAXABLE_BALANCE+B34)/2
write C36 'Mid-year avg: return × (start + end) / 2

write A37 'Tax-Def End
write B37 =B28+RETURN_RATE*(B9+B28)/2
write C37 'Start = after RMD, end = after withdrawals

write A38 'Tax-Free End
write B38 =B29+RETURN_RATE*(TAX_FREE_BALANCE+B29)/2

# Row 40: Results
write A40 '─── END OF YEAR ───

write A41 'Ending Taxable
write B41 =B36

write A42 'Ending Basis
write B42 =B35*(1-INFLATION_RATE)
write C42 'Basis eroded by inflation

write A43 'Ending Tax-Def
write B43 =B37

write A44 'Ending Tax-Free
write B44 =B38

write A45 'Total Taxes
write B45 =B13+B30

write A46 'Shortfall
write B46 =B31
write C46 '0 = success

# Outputs
output B41
output B42
output B43
output B44
output B45
output B46

# Formatting
default-format currency 0
format RETURN_RATE percent 1
format INFLATION_RATE percent 1
