# DPR_CALCULATOR — D&D 5e
# Expected damage per round for weapon attacks.
# Returns: DPR, hit_chance, crit_chance, damage_per_hit, damage_per_crit
# Handles advantage/disadvantage, expanded crit ranges, multiple attacks.

new DPR_CALCULATOR

input ATTACK_BONUS 5
input TARGET_AC 15
input DIE_SIZE 8
input DAMAGE_BONUS 3
input NUM_ATTACKS 1
input CRIT_RANGE 20
input ADV_MODE 0

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: TITLE
# ═══════════════════════════════════════════════════════════════════
write A1 'DPR CALCULATOR — D&D 5e
write A2 'Average damage per round using probability math (not dice rolls).
write A3 'Enter your die size (e.g. 8 for d8) — the function computes the average for you.
style A1 bold
style A1 fontsize 24

# ═══════════════════════════════════════════════════════════════════
# ROW 5-12: YOUR ATTACK (inputs)
# ═══════════════════════════════════════════════════════════════════
write A5 '─── YOUR ATTACK ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray
write A6 'Attack Bonus
write B6 =ATTACK_BONUS
write C6 'Total modifier to d20 (ability mod + proficiency + magic)
write A7 'Target AC
write B7 =TARGET_AC
write C7 'Armor Class to meet or exceed
write A8 'Damage Die Size
write B8 =DIE_SIZE
write C8 '4=dagger  6=shortsword  8=longsword  10=pike  12=greataxe
write A9 'Damage Bonus
write B9 =DAMAGE_BONUS
write C9 'Ability modifier added to damage (Strength or Dexterity)
write A10 'Number of Attacks
write B10 =NUM_ATTACKS
write C10 'Weapon attacks per round
write A11 'Crit Range
write B11 =CRIT_RANGE
write C11 'Minimum d20 roll for critical (usually 20)
write A12 'Advantage Mode
write B12 =ADV_MODE
write C12 '0=Normal  1=Advantage  2=Disadvantage

# ═══════════════════════════════════════════════════════════════════
# ROW 14-18: HIT PROBABILITY
# ═══════════════════════════════════════════════════════════════════
write A14 '─── HIT PROBABILITY ───
highlight A14 gray
highlight B14 gray
highlight C14 gray
highlight D14 gray
highlight E14 gray
highlight F14 gray
highlight G14 gray
write A15 'Base Hit Chance
write B15 =MAX(0.05,MIN(0.95,(21+ATTACK_BONUS-TARGET_AC)/20))
write C15 '(21 + bonus - AC) / 20, clamped [5%, 95%]
write A16 'Hit Chance
write B16 =IF(ADV_MODE=1,1-(1-B15)^2,IF(ADV_MODE=2,B15^2,B15))
write C16 'Adjusted for advantage or disadvantage
write A17 'Base Crit Chance
write B17 =MAX(0,MIN(1,(21-CRIT_RANGE)/20))
write C17 '(21 - crit range) / 20
write A18 'Crit Chance
write B18 =IF(ADV_MODE=1,1-(1-B17)^2,IF(ADV_MODE=2,B17^2,B17))
write C18 'Adjusted for advantage or disadvantage

# ═══════════════════════════════════════════════════════════════════
# ROW 20-22: DAMAGE BREAKDOWN
# ═══════════════════════════════════════════════════════════════════
write A20 '─── DAMAGE BREAKDOWN ───
highlight A20 gray
highlight B20 gray
highlight C20 gray
highlight D20 gray
highlight E20 gray
highlight F20 gray
highlight G20 gray
write A21 'Average Die Roll
write B21 =(DIE_SIZE+1)/2
write C21 '(die size + 1) / 2 — e.g. d8 → 4.5
write A22 'Damage per Hit
write B22 =B21+DAMAGE_BONUS
write C22 'Average die roll + damage bonus
write A23 'Damage per Crit
write B23 =2*B21+DAMAGE_BONUS
write C23 '2x average die roll + bonus (only dice double in 5e)

# ═══════════════════════════════════════════════════════════════════
# ROW 25-31: RESULT
# ═══════════════════════════════════════════════════════════════════
write A25 '─── RESULT ───
highlight A25 gray
highlight B25 gray
highlight C25 gray
highlight D25 gray
highlight E25 gray
highlight F25 gray
highlight G25 gray
write A26 'DPR per Attack
write B26 =(B16-B18)*B22+B18*B23
write C26 '(hit - crit) x hit dmg + crit x crit dmg
write A27 'Total DPR
write B27 =B26*NUM_ATTACKS
write C27 '← OUTPUT
highlight B27 yellow
write A28 'Hit Chance
write B28 =B16
write C28 '← OUTPUT
write A29 'Crit Chance
write B29 =B18
write C29 '← OUTPUT
write A30 'Damage per Hit
write B30 =B22
write C30 '← OUTPUT
write A31 'Damage per Crit
write B31 =B23
write C31 '← OUTPUT

# Named outputs
name B27 DPR
name B28 HIT_CHANCE
name B29 CRIT_CHANCE
name B30 DAMAGE_PER_HIT
name B31 DAMAGE_PER_CRIT

# Outputs
output DPR
output HIT_CHANCE
output CRIT_CHANCE
output DAMAGE_PER_HIT
output DAMAGE_PER_CRIT

# Formatting
default-format number 2
format B6 number 0
format B7 number 0
format B8 number 0
format B9 number 0
format B10 number 0
format B11 number 0
format B12 number 0
format B15 percent 1
format B16 percent 1
format B17 percent 1
format B18 percent 1
format B28 percent 1
format B29 percent 1

# Test cases
# Basic: +5 vs AC 15, d8+3, normal
test "5, 15, 8, 3, 1, 20, 0" "4.35, 0.55, 0.05, 7.5, 12"
# Advantage
test "5, 15, 8, 3, 1, 20, 1" "6.42, 0.7975, 0.0975, 7.5, 12"
# Disadvantage
test "5, 15, 8, 3, 1, 20, 2" "2.28, 0.3025, 0.0025, 7.5, 12"
# Two attacks
test "5, 15, 8, 3, 2, 20, 0" "8.7, 0.55, 0.05, 7.5, 12"
# Champion crit 19-20
test "5, 15, 8, 3, 1, 19, 0" "4.575, 0.55, 0.1, 7.5, 12"
# Easy target AC 10
test "5, 10, 8, 3, 1, 20, 0" "6.225, 0.8, 0.05, 7.5, 12"
# No damage bonus (d6 only)
test "5, 15, 6, 0, 1, 20, 0" "2.1, 0.55, 0.05, 3.5, 7"
# Nearly impossible (nat 20 only)
test "0, 25, 8, 3, 1, 20, 0" "0.6, 0.05, 0.05, 7.5, 12"

save
