# FISHERINV - Inverse Fisher z-transformation
# Converts Fisher z back to a correlation coefficient
# Formula: r = (e^(2y) - 1) / (e^(2y) + 1)

new FISHERINV
input Y 0.9730

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'INVERSE FISHER TRANSFORMATION
style A1 bold
style A1 fontsize 24
write A2 'Converts a Fisher z-score back to a correlation coefficient.
write A3 'r = (e^(2y) - 1) / (e^(2y) + 1)

# ═══════════════════════════════════════════════════════════════════
# ROW 5-6: INPUT
# ═══════════════════════════════════════════════════════════════════

write A5 '─── INPUT ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Fisher z
write B6 =Y
write C6 'The transformed value

# ═══════════════════════════════════════════════════════════════════
# ROW 8-12: THE INVERSE TRANSFORMATION
# ═══════════════════════════════════════════════════════════════════

write A8 '─── THE INVERSE TRANSFORMATION ───
highlight A8 gray
highlight B8 gray
highlight C8 gray
highlight D8 gray
highlight E8 gray
highlight F8 gray
highlight G8 gray

write A9 '2y
write B9 =2*B6
write C9 'Double the z-score

write A10 'e^(2y)
write B10 =EXP(B9)
write C10 'Exponential

write A11 'e^(2y) - 1
write B11 =B10-1
write C11 'Numerator

write A12 'e^(2y) + 1
write B12 =B10+1
write C12 'Denominator

# ═══════════════════════════════════════════════════════════════════
# ROW 14-15: RESULT
# ═══════════════════════════════════════════════════════════════════

write A14 '─── RESULT ───
highlight A14 gray
highlight B14 gray
highlight C14 gray
highlight D14 gray
highlight E14 gray
highlight F14 gray
highlight G14 gray

write A15 'Correlation (r)
write B15 =B11/B12
write C15 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B15 FISHERINV
output FISHERINV

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 4

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Inverse of FISHER(0.75) → 0.75
test "0.9729550745276566" "0.75"

# Zero (identity)
test "0" "0"

# Inverse of FISHER(0.5) → 0.5
test "0.5493061443340548" "0.5"

# Negative z → negative correlation
test "-0.30951960420311175" "-0.3"

# Large z → strong correlation (inverse of FISHER(0.9))
test "1.4722194895832204" "0.9"

save
