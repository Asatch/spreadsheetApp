# COMBIN - Combinations
# Returns the number of ways to choose k items from n (order doesn't matter)
# Formula: n! / (k! × (n-k)!)

new COMBIN
use FACTORIAL
input N 10
input K 3

# ═══════════════════════════════════════════════════════════════════
# ROW 1-3: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'COMBINATIONS
style A1 bold
style A1 fontsize 24
write A2 'Returns the number of combinations for a given number of items.
write A3 'Order does not matter: choosing A,B,C is the same as C,B,A.

# ═══════════════════════════════════════════════════════════════════
# ROW 5-8: THE QUESTION
# ═══════════════════════════════════════════════════════════════════

write A5 '─── THE QUESTION ───
highlight A5 gray
highlight B5 gray
highlight C5 gray
highlight D5 gray
highlight E5 gray
highlight F5 gray
highlight G5 gray

write A6 'Total Items (n)
write B6 =N
write C6 'The pool to choose from

write A7 'Items to Choose (k)
write B7 =K
write C7 'How many we select

write A8 'Items Not Chosen
write B8 =B6-B7
write C8 'n - k

# ═══════════════════════════════════════════════════════════════════
# ROW 10-13: COUNTING ARRANGEMENTS
# ═══════════════════════════════════════════════════════════════════

write A10 '─── COUNTING ARRANGEMENTS ───
highlight A10 gray
highlight B10 gray
highlight C10 gray
highlight D10 gray
highlight E10 gray
highlight F10 gray
highlight G10 gray

write A11 'n!
write B11 =FACTORIAL(B6)
write C11 'All possible orderings of n items

write A12 'k!
write B12 =FACTORIAL(B7)
write C12 'Orderings of chosen items (irrelevant)

write A13 '(n-k)!
write B13 =FACTORIAL(B8)
write C13 'Orderings of unchosen items (irrelevant)

# ═══════════════════════════════════════════════════════════════════
# ROW 15-16: THE FORMULA
# ═══════════════════════════════════════════════════════════════════

write A15 '─── THE FORMULA ───
highlight A15 gray
highlight B15 gray
highlight C15 gray
highlight D15 gray
highlight E15 gray
highlight F15 gray
highlight G15 gray

write A16 'k! × (n-k)!
write B16 =B12*B13
write C16 'Orderings we divide out

# ═══════════════════════════════════════════════════════════════════
# ROW 18-19: RESULT
# ═══════════════════════════════════════════════════════════════════

write A18 '─── RESULT ───
highlight A18 gray
highlight B18 gray
highlight C18 gray
highlight D18 gray
highlight E18 gray
highlight F18 gray
highlight G18 gray

write A19 'Combinations
write B19 =B11/B16
write C19 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B19 COMBIN
output COMBIN

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

default-format number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Standard: 10 choose 3
test "10, 3" "120"

# Small: 5 choose 2
test "5, 2" "10"

# Large: poker hand (52 choose 5)
test "52, 5" "2598960"

# Edge: choose nothing
test "5, 0" "1"

# Edge: choose all
test "5, 5" "1"

save
