# MINUTE - Extract minute from Excel serial date/time
# Time is the decimal portion of a serial number

new MINUTE
use INT
input SERIAL 45306.604687500
default-format number 6

# ═══════════════════════════════════════════════════════════════════
# ROW 1-2: DOCSTRING
# ═══════════════════════════════════════════════════════════════════

write A1 'MINUTE FROM TIME
style A1 bold
style A1 fontsize 24
write A2 'Extracts the minute (0-59) from an Excel serial date/time.

# ═══════════════════════════════════════════════════════════════════
# ROW 4-5: INPUT
# ═══════════════════════════════════════════════════════════════════

write A4 '─── INPUT ───
highlight A4 gray
highlight B4 gray
highlight C4 gray
highlight D4 gray
highlight E4 gray
highlight F4 gray
highlight G4 gray

write A5 'Serial Date/Time
write B5 =SERIAL
write C5 'Days + time fraction

# ═══════════════════════════════════════════════════════════════════
# ROW 7-11: THE CALCULATION
# ═══════════════════════════════════════════════════════════════════

write A7 '─── THE CALCULATION ───
highlight A7 gray
highlight B7 gray
highlight C7 gray
highlight D7 gray
highlight E7 gray
highlight F7 gray
highlight G7 gray

write A8 'Time Fraction
write B8 =MOD(B5,1)
write C8 'Decimal part (0-1)

write A9 'Hours (decimal)
write B9 =B8*24
write C9 'Time as hours

write A10 'Hour Remainder
write B10 =MOD(B9,1)
write C10 'Fractional hours

write A11 'Minutes (decimal)
write B11 =B10*60
write C11 'Remainder as minutes

# ═══════════════════════════════════════════════════════════════════
# ROW 13-14: RESULT
# ═══════════════════════════════════════════════════════════════════

write A13 '─── RESULT ───
highlight A13 gray
highlight B13 gray
highlight C13 gray
highlight D13 gray
highlight E13 gray
highlight F13 gray
highlight G13 gray

write A14 'Minute
write B14 =INT(B11)
write C14 '← OUTPUT

# ═══════════════════════════════════════════════════════════════════
# OUTPUTS
# ═══════════════════════════════════════════════════════════════════

name B14 MINUTE
output MINUTE

# ═══════════════════════════════════════════════════════════════════
# FORMATTING
# ═══════════════════════════════════════════════════════════════════

format B9 number 4
format B11 number 4
format B14 number 0

# ═══════════════════════════════════════════════════════════════════
# TEST CASES
# ═══════════════════════════════════════════════════════════════════

# Midnight (0 minutes)
test "0" "0"

# Noon (exact hour, 0 minutes)
test "0.5" "0"

# 2:30:45 PM (30 minutes)
test "45306.604687500" "30"

# 0:30 AM (30 minutes past midnight)
test "45306.020833333333333" "30"

# 11:59:59 PM (59 minutes)
test "45306.999988426" "59"

# 1:00 AM (exact hour)
test "0.041666667" "0"

# 0:15 AM (15 minutes)
test "0.010416667" "15"

save
